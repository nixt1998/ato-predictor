# Server
server = function(input, output, session) {
  
  # Calculated features
  tAs = reactive({
    as.numeric(input$iAs) + as.numeric(input$MMA) + as.numeric(input$DMA)
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  PMI = reactive({
    # input$update
    as.numeric(input$MMA) / as.numeric(input$iAs)
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  SMI = reactive({
    # input$update
    as.numeric(input$DMA) / as.numeric(input$MMA)
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  iAs_pct = reactive({
    # input$update
    as.numeric(input$iAs) / tAs() * 100
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  MMA_pct = reactive({
    # input$update
    as.numeric(input$MMA) / tAs() * 100
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  DMA_pct = reactive({
    # input$update
    as.numeric(input$DMA) / tAs() * 100
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  # Data used for prediction
  pred_data = reactive({
    tibble(tAs = tAs(),
           SMI = SMI(),
           MMA_per = MMA_pct(),
           DMA_per = DMA_pct(),
           CT_drug = input$ct_drug) |>
      mutate(CT_drug = factor(x = CT_drug, levels = c('No', 'Yes'))) |>
      bind_cols(non_select_features_data)
  }) |>
    bindEvent(input$update,
              ignoreNULL = FALSE,
              ignoreInit = FALSE)
  
  # Predicted class for input data
  .pred_class = reactive({
    optim_wflow |>
      predict(new_data = pred_data(), type = 'class') |>
      pull(.pred_class) |> 
      as.character()
  })
  
  # Predicted prob for input data
  .pred_Yes = reactive({
    optim_wflow_last_fit |>
      extract_workflow() |>
      predict(new_data = pred_data(), type = 'prob') |>
      pull(.pred_Yes)
  })
  
  # SHAP for input data
  optim_wflow_shap = reactive({
    kernelshap(object = optim_wflow,
               X = pred_data(),
               bg_X = train_data |> dplyr::select(-outcome),
               type = "prob",
               # parallel = TRUE,
               parallel_args = list(packages = "colino", "aorsf", "bonsai"))
  })
  
  shap_data = reactive({
    optim_wflow_shap()$S$.pred_Yes |>
      as_tibble() |>
      dplyr::select(all_of(select_features))
  })
  
  output$pred = renderUI({
    
    tags$div(
      div(style = "text-align: center;",
          tags$text(if_else(.pred_class() == 'No', 'No Cardiotoxicity', 'Cardiotoxicity'),
                    style = str_c("font-weight: bold; font-size: 30px; color: ", if_else(.pred_class() == 'No', 'green', 'red')))),
      
      div(style = "text-align: center;",
          tags$text(str_c('Risk: ', round(.pred_Yes() * 100, 2), '%'),
                    style = "font-weight: bold; font-size: 30px;"))
